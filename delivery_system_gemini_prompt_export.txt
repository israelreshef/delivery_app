אפשרות שינוי סטטוס עוסק בהגדרות המערכת
קשירה אוטומטית ללקוח מכרטיס הלקוח
מספור אוטומטי רציף (לפי חוק)
הפקה כ-PDF ושליחה במייל/וואטסאפ

חלק ד' — Schema מסד הנתונים (PostgreSQL)

העתק לגמיני — זהו ה-Schema המוצע להוסיף/לשדרג:

טבלאות חדשות נדרשות


SQL חשוב — customers
CREATE TABLE customers (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  business_name VARCHAR(255) NOT NULL,
  tax_id VARCHAR(20) UNIQUE NOT NULL, -- ח.פ או ת.ז
  customer_type VARCHAR(20) NOT NULL, -- 'business' | 'individual'
  vat_status VARCHAR(20), -- 'exempt' | 'registered' | 'none'
  address TEXT,
  city VARCHAR(100),
  phone VARCHAR(20),
  email VARCHAR(255),
  payment_terms VARCHAR(50), -- 'immediate' | 'net30' | 'net60'
  price_tier VARCHAR(20) DEFAULT 'standard',
  discount_percent DECIMAL(5,2) DEFAULT 0,
  notes TEXT,
  status VARCHAR(20) DEFAULT 'active',
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

SQL חשוב — invoices
CREATE TABLE invoices (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  invoice_number SERIAL UNIQUE NOT NULL,
  document_type VARCHAR(30) NOT NULL,
  -- 'invoice' | 'receipt' | 'invoice_receipt' | 'credit_note'
  customer_id UUID REFERENCES customers(id),
  issue_date DATE NOT NULL DEFAULT CURRENT_DATE,
  due_date DATE,
  subtotal DECIMAL(12,2) NOT NULL,
  vat_amount DECIMAL(12,2) DEFAULT 0,
  total DECIMAL(12,2) NOT NULL,
  vat_rate DECIMAL(5,2) DEFAULT 0, -- 17 לעוסק מורשה, 0 לפטור
  status VARCHAR(20) DEFAULT 'draft',
  -- 'draft' | 'sent' | 'paid' | 'cancelled'
  pdf_url TEXT,
  notes TEXT,
  created_by UUID, -- courier/admin
  created_at TIMESTAMPTZ DEFAULT NOW()
);
